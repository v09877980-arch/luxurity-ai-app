# --- 2. USAGE LIMITS & RESET ENGINE ---
def get_today_date():
    return datetime.now().strftime("%Y-%m-%d")

def load_usage():
    if os.path.exists(USAGE_FILE):
        try:
            with open(USAGE_FILE, "r", encoding="utf-8") as f:
                data = json.load(f)
                if data.get("date") == get_today_date():
                    return data
        except Exception:
            pass
    return {
        "date": get_today_date(),
        "pro_used": 0,
        "max_used": 0,
        "img_used": 0,
        "tier": "Free"
    }

def save_usage(usage_data):
    try:
        with open(USAGE_FILE, "w", encoding="utf-8") as f:
            json.dump(usage_data, f, ensure_ascii=False, indent=2)
    except Exception:
        pass

if "usage" not in st.session_state:
    st.session_state.usage = load_usage()

# Check and reset daily limits at midnight
if st.session_state.usage["date"] != get_today_date():
    st.session_state.usage = {
        "date": get_today_date(),
        "pro_used": 0,
        "max_used": 0,
        "img_used": 0,
        "tier": st.session_state.usage.get("tier", "Free")
    }
    save_usage(st.session_state.usage)

def load_memory():
    if os.path.exists(MEMORY_FILE):
        try:
            with open(MEMORY_FILE, "r", encoding="utf-8") as f:
                return json.load(f)
        except Exception:
            return []
    return []

def save_memory(messages):
    try:
        with open(MEMORY_FILE, "w", encoding="utf-8") as f:
            json.dump(messages, f, ensure_ascii=False, indent=2)
    except Exception:
        pass

# --- 9. INPUT & LIMIT ENFORCEMENT ---
chat_input = st.chat_input("Message Luxurity AI...")
active_prompt = st.session_state.quick_prompt or chat_input
st.session_state.quick_prompt = None

if active_prompt or voice_recording:
    if not api_key:
        st.warning("⚠️ Please enter your Gemini API Key in the left sidebar!")
    else:
        # Enforce limits based on tier
        current_tier = st.session_state.usage["tier"]
        # Default search is unlimited for all plans. Let's count as pro/max if user requests heavy search/generation.
        if current_tier == "Free" and st.session_state.usage["pro_used"] >= 5:
            st.error("⚠️ Daily Pro search limit reached (5/day)! Upgrade to Luxurity Pro or Max for unlimited access.")
        else:
            if current_tier == "Free":
                st.session_state.usage["pro_used"] += 1
                save_usage(st.session_state.usage)

            display_text = active_prompt if active_prompt else "🎙️ [Voice Audio Query]"
            st.session_state.messages.append({"role": "user", "content": display_text})
            with st.chat_message("user"):
                st.markdown(display_text)

            with st.chat_message("assistant"):
                placeholder = st.empty()
                full_reply = ""

                try:
                    client = genai.Client(api_key=api_key)
                    tools = [{"google_search": {}}] if enable_search else None

                    contents = []
                    for m in st.session_state.messages[:-1]:
                        role = "user" if m["role"] == "user" else "model"
                        contents.append(types.Content(role=role, parts=[types.Part.from_text(text=m["content"])]))

                    current_parts = [types.Part.from_text(text=active_prompt if active_prompt else "Process voice input.")]
                    if voice_recording and not active_prompt:
                        current_parts.append(types.Part.from_bytes(data=voice_recording.getvalue(), mime_type="audio/wav"))

                    contents.append(types.Content(role="user", parts=current_parts))

                    response = client.models.generate_content_stream(
                        model="gemini-2.5-flash",
                        contents=contents,
                        config=types.GenerateContentConfig(
                            system_instruction=base_instruction,
                            tools=tools
                        )
                    )

                    for chunk in response:
                        if chunk.text:
                            full_reply += chunk.text
                            placeholder.markdown(full_reply + " ▌")

                    placeholder.markdown(full_reply)
                    st.session_state.messages.append({"role": "assistant", "content": full_reply})
                    save_memory(st.session_state.messages)

                    if enable_voice and full_reply:
                        try:
                            clean_speech = full_reply.replace("*", "").replace("#", "")[:500]
                            tts = gTTS(text=clean_speech, lang=lang_codes[language_choice])
                            audio_fp = BytesIO()
                            tts.write_to_fp(audio_fp)
                            st.audio(audio_fp, format="audio/mp3")
                        except Exception:
                            pass

                except Exception as e:
                    st.session_state.messages.pop()
                    st.error(f"❌ Error: {e}")
